# RMS SOW ↔ BRS Traceability Mapping — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add bidirectional traceability between the RMS SOW deliverables and the RMS BRS requirement IDs, surfaced inline + as an appendix in both `RMS_SOW.docx` and `RMS_TSD.docx`.

**Architecture:** A new pure-Python data module `_brs_mapping.py` (sibling to `_generate_sow_tsd.py`) holds the BRS catalogue, the deliverable→Req-ID map, and an out-of-scope index. The existing generator imports it, prepends a coverage assertion to both `build_sow()` and `build_tsd()`, adds a `BRS Refs` column to the three Phase deliverable tables, extends `feature_block()` with a `deliverable_id` parameter that renders a `BRS coverage:` line, and appends a new `§13. BRS Traceability Matrix` (forward + reverse) to both documents.

**Tech Stack:** Python 3.10+, `python-docx`, pytest. No new runtime dependencies.

**Source spec:** [docs/superpowers/specs/2026-05-09-rms-sow-brs-mapping-design.md](../specs/2026-05-09-rms-sow-brs-mapping-design.md)

**Project rule (overrides default):** This codebase prohibits auto-commits. Each task ends with **"Pause for user instruction"** instead of `git commit`. The user will say `commit` (or similar) before any commit is created.

```python
"""
BRS ↔ SOW traceability data for the RMS programme.

Source of truth for:
  - BRS_CATALOGUE       — every BRS Req-ID with its section and verbatim text
  - DELIVERABLE_BRS     — SOW deliverable -> [Req-IDs it satisfies]
  - OUT_OF_SCOPE_REQS   — Req-IDs explicitly excluded, with reason

Run `python roadmap/logistics/pricing/_brs_mapping.py` to self-check coverage.
"""
from __future__ import annotations

from typing import TypedDict


class BrsReq(TypedDict):
    id: str          # e.g. "RM-001"
    section: str     # e.g. "6.1 Rate Master Management"
    text: str        # verbatim from RMS.docx.pdf


# Populated in Task 2.
BRS_CATALOGUE: list[BrsReq] = []

# Populated in Task 3.
DELIVERABLE_BRS: dict[str, list[str]] = {}

# Populated in Task 3.
OUT_OF_SCOPE_REQS: dict[str, str] = {}


def reverse_index() -> dict[str, str | None]:
    """Return req_id -> deliverable_id (or None for out-of-scope)."""
    index: dict[str, str | None] = {}
    for deliverable_id, req_ids in DELIVERABLE_BRS.items():
        for rid in req_ids:
            index[rid] = deliverable_id
    for rid in OUT_OF_SCOPE_REQS:
        index.setdefault(rid, None)
    return index


def coverage_status(req_id: str) -> tuple[str, str]:
    """Return ('In scope', deliverable_id) or ('Out of scope', reason)."""
    for deliverable_id, req_ids in DELIVERABLE_BRS.items():
        if req_id in req_ids:
            return ("In scope", deliverable_id)
    if req_id in OUT_OF_SCOPE_REQS:
        return ("Out of scope", OUT_OF_SCOPE_REQS[req_id])
    return ("Unmapped", "")


def assert_full_coverage() -> None:
    """Every BRS_CATALOGUE id must be mapped or explicitly out-of-scope.

    Raise RuntimeError listing all gaps, unknown in-scope IDs, and
    unknown out-of-scope IDs. This is the single check that prevents
    silent drift between the BRS and the generated SOW/TSD.
    """
    catalogue_ids = {req["id"] for req in BRS_CATALOGUE}
    mapped = {rid for rids in DELIVERABLE_BRS.values() for rid in rids}
    out_of_scope = set(OUT_OF_SCOPE_REQS.keys())

    unmapped = catalogue_ids - mapped - out_of_scope
    unknown_in_scope = mapped - catalogue_ids
    unknown_oos = out_of_scope - catalogue_ids

    errors: list[str] = []
    if unmapped:
        errors.append(
            f"Unmapped BRS IDs (add to DELIVERABLE_BRS or OUT_OF_SCOPE_REQS): {sorted(unmapped)}"
        )
    if unknown_in_scope:
        errors.append(
            f"DELIVERABLE_BRS references IDs not in BRS_CATALOGUE: {sorted(unknown_in_scope)}"
        )
    if unknown_oos:
        errors.append(
            f"OUT_OF_SCOPE_REQS references IDs not in BRS_CATALOGUE: {sorted(unknown_oos)}"
        )
    if errors:
        raise RuntimeError("BRS coverage check failed:\n  - " + "\n  - ".join(errors))


if __name__ == "__main__":
    assert_full_coverage()
    print(f"OK: {len(BRS_CATALOGUE)} catalogue entries, "
          f"{sum(len(v) for v in DELIVERABLE_BRS.values())} mapped, "
          f"{len(OUT_OF_SCOPE_REQS)} out-of-scope.")
```

```python
    for d_id, req_ids in brs_mapping.DELIVERABLE_BRS.items():
```

```python
def _add_brs_traceability_section(doc: Document, *, deliverable_titles: dict[str, str]) -> None:
    """Append §13 BRS Traceability Matrix to a SOW or TSD doc.

    Two sub-sections:
      §13.1 Forward map — per-phase deliverable -> BRS Refs
      §13.2 Reverse map — every BRS Req-ID -> phase + deliverable + status
    """
    add_page_break(doc)
    add_heading(doc, "13. BRS Traceability Matrix", level=1)
    add_paragraph(
        doc,
        "Bidirectional map between SOW deliverables and BRS Req-IDs. "
        "Source: roadmap/logistics/pricing/RMS.docx.pdf and the mapping data "
        "in roadmap/logistics/pricing/_brs_mapping.py.",
        italic=True, color=MUTED,
    )

    # ── §13.1 Forward map: one table per phase ─────────────────────────────
    add_heading(doc, "13.1 Forward Map (Deliverable → BRS Refs)", level=2)
    for phase_label, phase_prefix in (("Phase 1", "P1-"), ("Phase 2", "P2-"), ("Phase 3", "P3-")):
        add_heading(doc, phase_label, level=3)
        rows: list[list[str]] = []
        for d_id in sorted(_brs_mapping.DELIVERABLE_BRS):
            if not d_id.startswith(phase_prefix):
                continue
            refs = _brs_mapping.DELIVERABLE_BRS[d_id]
            rows.append([
                d_id,
                deliverable_titles.get(d_id, ""),
                f"{len(refs)}",
                ", ".join(refs) if refs else "—",
            ])
        add_table(
            doc,
            headers=["#", "Feature", "Count", "BRS Refs"],
            rows=rows,
            col_widths_cm=[1.5, 5.5, 1.5, 7.5],
        )

    # ── §13.2 Reverse map: one row per Req-ID, grouped by BRS section ──────
    add_heading(doc, "13.2 Reverse Map (BRS Req-ID → Deliverable)", level=2)
    current_section: str | None = None
    rows = []
    for req in _brs_mapping.BRS_CATALOGUE:
        if req["section"] != current_section:
            if rows:
                add_table(
                    doc,
                    headers=["Req ID", "Requirement", "Phase", "Deliverable", "Status"],
                    rows=rows,
                    col_widths_cm=[2.0, 7.0, 1.5, 2.0, 3.5],
                )
                rows = []
            add_heading(doc, f"§{req['section']}", level=3)
            current_section = req["section"]
        status, target = _brs_mapping.coverage_status(req["id"])
        phase = target.split("-")[0] if status == "In scope" else "—"
        deliverable = target if status == "In scope" else "—"
        text = req["text"] if len(req["text"]) <= 80 else req["text"][:77] + "..."
        rows.append([req["id"], text, phase, deliverable, status])
    if rows:
        add_table(
            doc,
            headers=["Req ID", "Requirement", "Phase", "Deliverable", "Status"],
            rows=rows,
            col_widths_cm=[2.0, 7.0, 1.5, 2.0, 3.5],
        )
```
